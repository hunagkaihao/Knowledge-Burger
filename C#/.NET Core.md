# 发布说明

部署模式

1. 依赖框架：发布的主机需要部署相关的.net 运行时环境
2. 独立：将.net 运行时环境打包到发布文件中
3. ReadyToRun：程序运行时快，但编译时慢
4. 生成单个文件：生成极少文件，适合用于发布可运行包
5. 剪裁未使用的程序集：删去用不到的程序集然后打包

```C#
using System;
using System.Text;
using Microsoft.Extensions.DependencyInjection;

namespace TEST
{
    class Test
    {
        public static async Task Main()
        {
            ServiceCollection services = new ServiceCollection();
            services.AddTransient<TestServiceImp1>();
            using (ServiceProvider serviceProvider = services.BuildServiceProvider())
            {
                TestServiceImp1 testService = serviceProvider.GetService<TestServiceImp1>();
                testService.Name = "Tom";
                testService.SayHi();
            }
        }
        
    }

    public interface ITestService
    {
        public string Name { get; set; }
        public void SayHi();
    }
    public class TestServiceImp1 : ITestService
    {
        public string Name { get; set; }
        public void SayHi()
        {
            Console.WriteLine($"Hi,I'm {Name}");
        }
    }
    public class TestServiceImp2 : ITestService
    {
        public string Name { get; set; }
        public void SayHi()
        {
            Console.WriteLine($"你好， 我是{Name}");
        }
    }
}
```

```c#
using System;
using System.Collections.ObjectModel;
using System.Text;
using Microsoft.Extensions.DependencyInjection;

namespace TEST
{
    class Test
    {
        static void Main(string[] args)
        {
            ServiceCollection s = new ServiceCollection();
            s.AddScoped<Controller>();
            s.AddScoped<ILog, LogImpl>();
            //s.AddScoped<IConfig, ConfigImpl>();
            s.AddScoped<IConfig, DBConfigImpl>();
            s.AddScoped<IStorage, StorageImpl>();
            using (var scope = s.BuildServiceProvider())
            {
                var c = scope.GetRequiredService<Controller>();
                c.Test();
            }

        }
        
    }
    class Controller
    {
        public readonly ILog log;
        public readonly IStorage storage;
        public Controller(ILog log, IStorage storage)
        {
            this.log = log;
            this.storage = storage;
        }
        public void Test()
        {
            log.Log("开始上传");
            this.storage.Save("1.text", "sfasaada");
            log.Log("上传成功");
        }
    }
    public interface ILog
    {
        public void Log(string message);
    }
    public class LogImpl : ILog
    {
        public void Log(string mes)
        {
            Console.WriteLine($"日志；{mes}");
        }
    }
    public interface IConfig
    {
        public string GetValue(string name);
    }
    public class ConfigImpl : IConfig
    {
        public string GetValue(string name)
        {
            Console.WriteLine($"获取{name}配置");
            return name;
        }
    }
    public class DBConfigImpl : IConfig
    {
        public string GetValue(string name)
        {
            Console.WriteLine($"获取{name}数据库配置");
            return name;
        }
    }

    public interface IStorage
    {
        public void Save(string path, string value);
    }   
    public class StorageImpl : IStorage
    {
        public readonly IConfig config;
        public StorageImpl(IConfig config)
        {
            this.config = config;
        }
        public void Save(string path, string value)
        {
            string service = config.GetValue(path);
            Console.WriteLine($"向服务器{service}写入{value}");
        }
    }
}
```
